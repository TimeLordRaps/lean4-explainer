  | generator {X : C} : S X → Thick S X
  | zero : Thick S 0
  | iso {X Y : C} : (X ≅ Y) → Thick S X → Thick S Y

  | extension (T : Triangle C) : T ∈ distTriang C → Thick S T.obj₁ → Thick S T.obj₃ → Thick S T.obj₂
