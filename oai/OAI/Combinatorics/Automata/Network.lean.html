    def edgeInc (n : ℕ) : Edge Q n × Bool → WholeInc Q n
  | (.inl (j,slot),b) =>
      if slot.1 ^^ b then .inl (j.castSucc, .inl (true,slot))
      else .inl (j.succ, .inl (false,slot))
  | (.inr (.inl (i,pq)),b) => .inl (i,.inr (pq,b))
  | (.inr (.inr (s,slot)),b) =>
      if b then .inl (outCell n s,.inl (s,slot)) else .inr (s,slot)
