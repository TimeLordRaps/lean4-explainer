    def pi : (k : ℕ) → (A : Fin k → Type) → [∀ i, Fintype (A i)] →
    (∀ i, SamplerCoupling (A i)) → SamplerCoupling (∀ i, A i)
  | 0, _, _, _ => SamplerCoupling.pure (fun i => Fin.elim0 i)
  | k + 1, A, _, S =>
      ((S 0).prod (pi k (fun i => A i.succ) (fun i => S i.succ))).map (Fin.consEquiv A)
