    def onSlice {s : ℕ} : {j m : ℕ} → Slice s j → Nodes s m → Bool
  | _, 0, _, v => nomatch v
  | 0, _ + 1, .root, .inl _ => true
  | 0, _ + 1, .root, .inr _ => false
  | _ + 1, _ + 1, .next _ _, .inl _ => false
  | _ + 1, _ + 1, .next g child, .inr ((g', b), v) =>
      if g' = g then onSlice (child b) v else false
