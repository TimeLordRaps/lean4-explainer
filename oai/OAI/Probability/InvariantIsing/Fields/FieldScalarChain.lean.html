    def fieldScalarSquares : (L : List (ℝ × ℝ≥0)) → (ℝ → ℝ) → (ℝ → ℝ) →

  | [], _, a => fun _ z => (a z) ^ 2
  | av :: L, F, a => Fin.cons (fun z => (fieldScalarMean (av :: L) F a z) ^ 2)
