    def reflectionTerms {N : ℕ} : List (Count N) → List (ℝ × Count N)
  | [] => [(1,Count.empty)]
  | A::l => reflectionTerms l ++ (reflectionTerms l).map (fun p => (-2*p.1,A.join p.2))
