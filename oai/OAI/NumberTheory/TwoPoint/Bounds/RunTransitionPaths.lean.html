    def scanRunTransitions (runLabel : ℕ → ι) : ℕ → ℕ → ι → List (ℕ × ι)
  | _, 0, _ => []
  | n, len + 1, previous =>
      if runLabel n = previous then scanRunTransitions runLabel (n + 1) len previous
      else (n, runLabel n) :: scanRunTransitions runLabel (n + 1) len (runLabel n)
