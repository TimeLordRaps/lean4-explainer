    def Nat.iteratedLog (b : ℕ) : ℕ → ℕ
  | 0 => 0
  | 1 => 0
  | (n + 2) => Nat.iteratedLog b (Nat.log b (n + 2)) + 1
termination_by n => n
decreasing_by exact Nat.log_lt_self b (by omega)
