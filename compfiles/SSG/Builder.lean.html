    def copyAsset (outDir : String) : Asset → IO Unit
  | .file src dst => do
    let content ← IO.FS.readFile src
    IO.FS.writeFile (outDir ++ "/" ++ dst) content
  | .binary src dst => do
    let content ← IO.FS.readBinFile src
    IO.FS.writeBinFile (outDir ++ "/" ++ dst) content
